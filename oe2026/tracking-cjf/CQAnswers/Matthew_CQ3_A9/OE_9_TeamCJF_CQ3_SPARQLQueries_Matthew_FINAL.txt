# QUERY A1 — Which conversation moves explicitly support Platform A?
# Reasoning: none. Reads source/target links of Support relationship individuals.
# Observed with Reasoner=None: 6 results (all Claims):
#   M0004, M0008, M0018, M0019, M0038, M0060.

PREFIX cjf: <https://tw.rpi.edu/ontology-engineering/oe2026/tracking-cjf/>

SELECT DISTINCT ?claim
WHERE {
    ?claim cjf:makesSupports ?relationship .
    ?relationship cjf:isAimedAtSupports cjf:OCT21_A-M0003 .
}
ORDER BY ?claim


# QUERY A2 — Which conversation moves explicitly challenge Platform A?

# QUERY B — Are there any Claims with a currently recorded SupportedContested state?
# Show the recorded state INDIVIDUAL and the Claim/Issue that challenged it.
# Reasoning: none. Uses an exact rdfs:comment annotation on the *current*
# StateAssignment to select the annotated reference state; this does NOT
# establish OWL inference of rdf:type cjf:SupportedContested.
# Observed with Reasoner=None: 1 row:
#   ?claim      = cjf:OCT21_A-M0023
#   ?state      = cjf:OCT21_A-M0023_StateAtTurn12
#   ?challenger = cjf:OCT21_A-M0028
# Scope: tests the currently annotated SupportedContested case. It does NOT
# exhaustively detect every possible inferred Contested or SupportedContested Claim.

PREFIX cjf: <https://tw.rpi.edu/ontology-engineering/oe2026/tracking-cjf/>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?claim ?state ?challenger
WHERE {
    ?claim rdf:type cjf:Claim .
    ?claim cjf:hasCurrentStateAssignment ?assignment .

    ?assignment rdfs:comment "Annotated reference state: SupportedContested" .
    ?assignment cjf:assignedState ?state .
    ?assignment cjf:hasTriggeringRelationship ?relationship .

    ?relationship rdf:type cjf:Challenge .
    ?relationship cjf:isAimedAtChallenges ?claim .

    ?challenger cjf:makesChallenges ?relationship .
}
ORDER BY ?claim


# QUERY C1 — Which Claims support Platform A AFTER OWL property-chain reasoning?
# Required inference: OWL 2 DL property chain
#    makesSupports o isAimedAtSupports  ->  supports
# Plugin/setup: Snap SPARQL Query with HermiT started in Protégé.
# Observed: 0 results with Reasoner=None; 6 with HermiT active:
#   M0004, M0008, M0018, M0019, M0038, M0060.
# This None-vs-HermiT comparison demonstrates the inferred direct relation.
# RDFS-only entailment is not sufficient for this OWL property-chain axiom.

PREFIX cjf: <https://tw.rpi.edu/ontology-engineering/oe2026/tracking-cjf/>

SELECT DISTINCT ?move
WHERE {
    ?move cjf:supports cjf:OCT21_A-M0003 .
}
ORDER BY ?move


# QUERY C2 — Which conversation moves challenge Platform A AFTER OWL reasoning?
# Required inference: OWL 2 DL property chain
#    makesChallenges o isAimedAtChallenges  ->  challenges
# Plugin/setup: Snap SPARQL Query with HermiT started in Protégé.
# Observed with HermiT active: 6 results:
#   M0010, M0013, M0014, M0015, M0016, M0050.

PREFIX cjf: <https://tw.rpi.edu/ontology-engineering/oe2026/tracking-cjf/>

SELECT DISTINCT ?move
WHERE {
    ?move cjf:challenges cjf:OCT21_A-M0003 .
